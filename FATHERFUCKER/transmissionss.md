## Causal Memory Engine (The Critical Fix from Transmission 4)

The engine must implement a history-dependent state, not just the current waypoint:


python
class CausalMemory:
    def __init__(self, tau: float = 8.0):
        self.tau = tau  # decay time constant
        self.trace = [0.0] * 6

    def update(self, target: list[float], dt: float) -> list[float]:
        """dm/dt = (d - m) / tau — first-order low-pass filter."""
        for i in range(6):
            self.trace[i] += (target[i] - self.trace[i]) * dt / self.tau
        return self.trace.copy()

    @property
    def lag(self) -> float:
        """Normalized lag between target and trace (0 = caught up, 1 = max lag)."""
        return 1.0 - math.exp(-dt / self.tau)  # approximate

class CausalMemory:
    def __init__(self, tau: float = 25.0):
        self.tau = tau
        self.trace = [0.0] * 6
        self.target = [0.0] * 6

    def update(self, target: list[float], dt: float) -> list[float]:
        self.target = target.copy()
        tau = max(self.tau, 1e-4)
        alpha = -math.expm1(-dt / tau)

        for i in range(6):
            self.trace[i] += alpha * (target[i] - self.trace[i])

        return self.trace.copy()

    @property
    def lag_vector(self) -> list[float]:
        return [
            self.target[i] - self.trace[i]
            for i in range(6)
        ]

    @property
    def lag_magnitude(self) -> float:
        return math.sqrt(
            sum(x * x for x in self.lag_vector) / 6.0
        )
